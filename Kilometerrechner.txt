"Wir wollen Meilen in Kilometer umrechen. "
"Eine Meile entspricht 1,60934 Kilometer. "
"Schreibe ein Programm"

MILE_TO_KILOMETER_FACTOR = 1.60934

input_miles = input("Geben Sie die Anzahl der Meilen ein: ")

miles = int(input_miles)
kilometers = miles * MILE_TO_KILOMETER_FACTOR
print(f"{miles} Meilen entsprechen {kilometers}.", kilometers, "km")
